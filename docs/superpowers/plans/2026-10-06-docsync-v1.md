# docsync v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `docsync` CLI exactly as defined by SPEC.md version 1: `docsync` checks, `docsync --write` records reviews.

**Architecture:** Small modules, one per spec area: `core` (records, ordering, quoting, diagnostics), `pattern` (§8), `universe` (§6, §7), `config` (§9), `hash` (§10), `lock` (§11), `engine` (§12, pure), `cli` (§13, §16), `report` (§14, §15). The engine receives a filesystem facade, so every state is unit-testable without disk. Tests are named by spec clause.

**Tech Stack:** TypeScript 7 strict, Node 24, ESM, pnpm, hand-written argument parsing (Task 12), `yaml` (only runtime dependency), tsup, vitest, oxlint, oxfmt, knip.

**Spec:** `docs/SPEC.md` (behavior, normative). Rationale: `docs/superpowers/specs/2026-10-06-docsync-design.md`. Read both before any task. When this plan and the spec disagree, the spec wins; report the conflict.

## Global Constraints

- Node.js `>=24.0.0`; `"type": "module"`; TypeScript strict; `tsc --noEmit` clean.
- No network, no git invocation, no clock, no `Intl`, no `localeCompare` in `src/` (SPEC §2, §3.3).
- Path order = UTF-16 code unit order: compare with `<` / `>` on strings, never `localeCompare`.
- All output and Lockfile strings escaped with `quote()` (SPEC §3.4) only.
- stdout/stderr: UTF-8, LF, final LF (SPEC §14.1).
- Exit codes 0/1/2/70 (SPEC §16).
- Runtime dependency: `yaml` only. Ask the user before adding any other dependency.
- Every source file implementing a clause has a `// SPEC §x.y` comment on the exported function. Test `describe` names start with the clause: `describe('§8.4 Select', ...)`.
- Max line length 100. Comments: max 3 lines.
- Commits: Conventional Commits. End each commit message with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Shell: one atomic command per call; no pipes, `&&`, or `$(...)`.

## Review Focus

1. A repo with no `.gitignore` and a huge `node_modules`: walk must still finish; only covered files are read (Task 6 test "does not read uncovered files").
2. Windows checkout with CRLF everywhere, including `docsync.lock`: verdict and lock unchanged (Task 8 CRLF test, Task 9 "tolerates CRLF lock" test).
3. A pattern written with a Windows backslash (`src\cli`): parses as literal, then fails as `E_EMPTY_PATTERN` instead of silently matching nothing (Task 10 test).
4. `docsync --write` run after a lock merge conflict: `E_LOCK`, nothing written, message names the fix (Task 13 e2e test).
5. A Dependent renamed on disk but not in config: `E_DEPENDENT_MISSING`, other Dependents still evaluated, exit 2 (Task 10 test).

```ts
// SPEC §3.3
export function comparePaths(a: string, b: string): number {
  if (a < b) return -1;
  return a > b ? 1 : 0;
}

export function sortPaths(xs: readonly string[]): string[] {
  return [...xs].sort(comparePaths);
}
```

`src/core/quote.ts`:
```ts
const NAMED: Record<string, string> = {
  '"': '\\"', '\\': '\\\\', '\b': '\\b', '\t': '\\t', '\n': '\\n', '\f': '\\f', '\r': '\\r',
};

function needsUnicodeEscape(cp: number): boolean {
  return cp <= 0x1f || (cp >= 0x7f && cp <= 0x9f) || cp === 0x2028 || cp === 0x2029 || cp === 0xfeff;
}

// SPEC §3.4
export function quote(s: string): string {
  let out = '"';
  for (const ch of s) {
    const named = NAMED[ch];
    const cp = ch.codePointAt(0) ?? 0;
    if (named !== undefined) out += named;
    else if (needsUnicodeEscape(cp)) out += `\\u${cp.toString(16).padStart(4, '0')}`;
    else out += ch;
  }
  return `${out}"`;
}

export function needsQuoting(s: string): boolean {
  return quote(s) !== `"${s}"`;
}
```

```ts
export function isRepoPath(s: string): boolean {
  if (s === '' || s.includes('\u0000') || s.normalize('NFC') !== s) return false;
  return s.split('/').every((seg) => seg !== '' && seg !== '.' && seg !== '..');
```

```ts
export function diag(
  code: Code,
  fields: { dependent?: string; subject?: string; message?: string } = {},
): Diagnostic {
  return {
    code,
    severity: code === 'W_ORPHAN' ? 'warning' : 'error',
    dependent: fields.dependent ?? '',
    subject: fields.subject ?? '',
    message: fields.message ?? FIX[code],
  };
}

// SPEC §5.5
export function sortDiagnostics(ds: readonly Diagnostic[]): Diagnostic[] {
  const sorted = [...ds].sort(
    (a, b) =>
      comparePaths(a.dependent, b.dependent) ||
      comparePaths(a.code, b.code) ||
      comparePaths(a.subject, b.subject),
  );
  return sorted.filter((d, i) => {
    const prev = sorted[i - 1];
    return !(prev && prev.code === d.code && prev.dependent === d.dependent &&
      prev.subject === d.subject && prev.message === d.message);
  });
}

export class Raised extends Error {
  constructor(readonly diagnostics: Diagnostic[]) {
    super(diagnostics.map((d) => d.code).join(', '));
  }
}
```

```ts
export type Atom =
  | { kind: 'lit'; value: string }
  | { kind: 'star' }
  | { kind: 'any' }
  | { kind: 'class'; negated: boolean; items: Array<[string, string]> }
  | { kind: 'alt'; alts: Atom[][] };
export type Segment = { kind: 'globstar' } | { kind: 'parts'; atoms: Atom[] };
export interface ParsedPattern { negated: boolean; segments: Segment[] }

class Invalid extends Error {}

class Reader {
  private i = 0;
  private readonly cps: string[];
  constructor(src: string) { this.cps = [...src]; }
  peek(): string | undefined { return this.cps[this.i]; }
  next(): string {
    const c = this.cps[this.i++];
    if (c === undefined) throw new Invalid();
    return c;
  }
  done(): boolean { return this.i >= this.cps.length; }
}

const LITERAL_STOP = new Set(['*', '?', '[', ']', '{', '}', ',', '/', '\\']);

function readEscape(r: Reader): string {
  const c = r.next();
  if (c === '/') throw new Invalid();
  return c;
}

function readClass(r: Reader): Atom {
  let negated = false;
  if (r.peek() === '!') { r.next(); negated = true; }
  const chars: string[] = [];
  while (r.peek() !== ']') {
    const c = r.next();
    if (c === '/') throw new Invalid();
    chars.push(c === '\\' ? '\\' + readEscape(r) : c);
  }
  r.next();
  if (chars.length === 0) throw new Invalid();
  const plain = (c: string) => (c.startsWith('\\') ? c.slice(1) : c);
  const items: Array<[string, string]> = [];
  for (let k = 0; k < chars.length; k++) {
    const c = chars[k]!;
    const isRange = chars[k + 1] === '-' && k + 2 < chars.length && k > -1;
    if (isRange && k + 1 !== chars.length - 1) {
      const lo = plain(c);
      const hi = plain(chars[k + 2]!);
      if ((lo.codePointAt(0) ?? 0) > (hi.codePointAt(0) ?? 0)) throw new Invalid();
      items.push([lo, hi]);
      k += 2;
    } else {
      items.push([plain(c), plain(c)]);
    }
  }
  return { kind: 'class', negated, items };
}

function readAtoms(r: Reader, inAlt: boolean, alternation: boolean): Atom[] {
  const atoms: Atom[] = [];
  for (;;) {
    const c = r.peek();
    if (c === undefined || c === '/') break;
    if (inAlt && (c === ',' || c === '}')) break;
    r.next();
    if (c === '*') {
      if (atoms.at(-1)?.kind === 'star') throw new Invalid();
      atoms.push({ kind: 'star' });
    } else if (c === '?') atoms.push({ kind: 'any' });
    else if (c === '[') atoms.push(readClass(r));
    else if (c === '{') {
      if (inAlt || !alternation) throw new Invalid();
      const alts: Atom[][] = [];
      for (;;) {
        const alt = readAtoms(r, true, alternation);
        if (alt.length === 0) throw new Invalid();
        alts.push(alt);
        const sep = r.next();
        if (sep === '}') break;
        if (sep !== ',') throw new Invalid();
      }
      if (alts.length < 2) throw new Invalid();
      atoms.push({ kind: 'alt', alts });
    } else if (c === '\\') atoms.push({ kind: 'lit', value: readEscape(r) });
    else if (LITERAL_STOP.has(c)) throw new Invalid();
    else atoms.push({ kind: 'lit', value: c });
  }
  return atoms;
}

function isDotSegment(atoms: Atom[]): boolean {
  const text = atoms.map((a) => (a.kind === 'lit' ? a.value : '\u0000')).join('');
  return text === '.' || text === '..';
}

// SPEC §8.1
export function parsePattern(
  src: string,
  opts: { alternation?: boolean } = {},
): ParsedPattern | null {
  if (src.normalize('NFC') !== src) return null;
  const alternation = opts.alternation ?? true;
  const negated = src.startsWith('!');
  const body = negated ? src.slice(1) : src;
  if (body === '' || body.startsWith('/') || body.endsWith('/')) return null;
  try {
    const segments: Segment[] = body.split('/').map((raw) => {
      if (raw === '**') return { kind: 'globstar' } as const;
      const r = new Reader(raw);
      const atoms = readAtoms(r, false, alternation);
      if (!r.done() || atoms.length === 0 || isDotSegment(atoms)) throw new Invalid();
      return { kind: 'parts', atoms } as const;
    });
    return { negated, segments };
  } catch (e) {
    if (e instanceof Invalid) return null;
    throw e;
  }
}
```

Note: splitting on `/` before parsing is safe because `/` is never legal inside a Class, Alternation or Escape (§8.1 grammar).

```ts
import { sortPaths } from '../core/order.ts';
import type { Atom, ParsedPattern, Segment } from './parse.ts';

const reEscape = (s: string) => s.replace(/[\\^$.*+?()[\]{}|/-]/gu, '\\$&');

function atomSource(a: Atom): string {
  switch (a.kind) {
    case 'lit': return reEscape(a.value);
    case 'star': return '[^/]*';
    case 'any': return '[^/]';
    case 'class': {
      const body = a.items
        .map(([lo, hi]) => (lo === hi ? reEscape(lo) : `${reEscape(lo)}-${reEscape(hi)}`))
        .join('');
      return a.negated ? `[^/${body}]` : `(?:(?!/)[${body}])`;
    }
    case 'alt': return `(?:${a.alts.map((alt) => alt.map(atomSource).join('')).join('|')})`;
  }
}

// SPEC §8.2
export function compileGlob(segments: Segment[]): RegExp {
  let src = '';
  segments.forEach((seg, i) => {
    const last = i === segments.length - 1;
    if (seg.kind === 'globstar') {
      src += last ? '[^/]+(?:/[^/]+)*' : '(?:[^/]+/)*';
      return;
    }
    src += seg.atoms.map(atomSource).join('') + (last ? '' : '/');
  });
  return new RegExp(`^${src}$`, 'u');
}

const cache = new WeakMap<ParsedPattern, RegExp>();
function regexOf(p: ParsedPattern): RegExp {
  let re = cache.get(p);
  if (!re) {
    re = compileGlob(p.segments);
    cache.set(p, re);
  }
  return re;
}

// SPEC §8.3
export function patternMatches(p: ParsedPattern, path: string): boolean {
  const re = regexOf(p);
  if (re.test(path)) return true;
  for (let i = path.indexOf('/'); i !== -1; i = path.indexOf('/', i + 1)) {
    if (re.test(path.slice(0, i))) return true;
  }
  return false;
}

// SPEC §8.4
export function select(patterns: readonly ParsedPattern[], candidates: readonly string[]): string[] {
  return sortPaths(candidates).filter((path) => {
    let selected = false;
    for (const p of patterns) if (patternMatches(p, path)) selected = !p.negated;
    return selected;
  });
}
```

```ts
import { parsePattern } from '../pattern/parse.ts';
import { compileGlob } from '../pattern/match.ts';

export interface IgnoreRule {
  scope: string;
  negated: boolean;
  dirOnly: boolean;
  anchored: boolean;
  re: RegExp;
}

function trimTrailingSpaces(line: string): string {
  let end = line.length;
  while (end > 0 && line[end - 1] === ' ' && line[end - 2] !== '\\') end--;
  return line.slice(0, end);
}

// SPEC §7.3
export function parseIgnoreLines(text: string, scope: string): IgnoreRule[] {
  const body = text.startsWith('﻿') ? text.slice(1) : text;
  const out: IgnoreRule[] = [];
  for (const rawLine of body.split('\n')) {
    let line = trimTrailingSpaces(rawLine.endsWith('\r') ? rawLine.slice(0, -1) : rawLine);
    if (line === '' || line.startsWith('#')) continue;
    let negated = false;
    if (line.startsWith('!')) { negated = true; line = line.slice(1); }
    let dirOnly = false;
    if (line.endsWith('/')) { dirOnly = true; line = line.slice(0, -1); }
    const anchored = line.includes('/');
    if (line.startsWith('/')) line = line.slice(1);
    const parsed = parsePattern(line.startsWith('!') ? `\\${line}` : line, { alternation: false });
    if (!parsed || parsed.negated) continue;
    out.push({ scope, negated, dirOnly, anchored, re: compileGlob(parsed.segments) });
  }
  return out;
}

function ruleMatches(rule: IgnoreRule, path: string): boolean {
  const prefix = rule.scope === '' ? '' : `${rule.scope}/`;
  if (!path.startsWith(prefix) || path === rule.scope) return false;
  const rel = path.slice(prefix.length);
  return rule.anchored ? rule.re.test(rel) : rule.re.test(rel.slice(rel.lastIndexOf('/') + 1));
}

// SPEC §7.3
export function isIgnored(path: string, isDir: boolean, rules: readonly IgnoreRule[]): boolean {
  let result = false;
  for (const rule of rules) {
    if ((!rule.dirOnly || isDir) && ruleMatches(rule, path)) result = !rule.negated;
  }
  return result;
}
```

Note on `\#`: `parsePattern` treats `\#` as an Escape of `#`, which is exactly the literal `#`. A leading `\!` likewise. A line starting with `!` after the negation (`!!a`) is passed as `\!a` so the inner `!` stays literal.

```ts
import { existsSync, readdirSync, readFileSync, statSync, type Dirent } from 'node:fs';
import { dirname, join, resolve } from 'node:path';
import { Raised, diag } from '../core/diagnostics.ts';
import { comparePaths } from '../core/order.ts';
import type { Config, Diagnostic } from '../core/types.ts';
import { isIgnored, parseIgnoreLines, type IgnoreRule } from './ignore.ts';

export type Kind = 'file' | 'link';
export interface Universe {
  paths: string[];
  kinds: Map<string, Kind>;
  onDisk: Map<string, string>;
}

// SPEC §6
export function determineRoot(cwd: string, rootOption: string | undefined): string {
  if (rootOption !== undefined) {
    const dir = resolve(cwd, rootOption);
    if (!existsSync(dir) || !statSync(dir).isDirectory()) throw new Raised([diag('E_ROOT')]);
    return dir;
  }
  let dir = cwd;
  for (;;) {
    if (existsSync(join(dir, 'docsync.yaml'))) return dir;
    const parent = dirname(dir);
    if (parent === dir) throw new Raised([diag('E_CONFIG_MISSING')]);
    dir = parent;
  }
}

const join2 = (p: string, n: string) => (p === '' ? n : `${p}/${n}`);
const isValidUtf8 = (name: Buffer) => Buffer.from(name.toString('utf8'), 'utf8').equals(name);

// SPEC §7.2
export function computeUniverse(root: string, config: Config): Universe {
  const errors: Diagnostic[] = [];
  const found: Array<{ path: string; kind: Kind }> = [];
  const configRules = config.ignore.flatMap((l) => parseIgnoreLines(l, ''));
  let rules: IgnoreRule[] = [];

  const walk = (abs: string, prefix: string): void => {
    let entries: Dirent<Buffer>[];
    try {
      entries = readdirSync(abs, { withFileTypes: true, encoding: 'buffer' });
    } catch {
      errors.push(diag('E_UNREADABLE', { subject: prefix }));
      return;
    }
    const before = rules;
    const gi = entries.find((e) => e.name.toString() === '.gitignore' && e.isFile());
    if (config.useGitignore && gi) {
      try {
        rules = [...rules, ...parseIgnoreLines(readFileSync(join(abs, '.gitignore'), 'utf8'), prefix)];
      } catch {
        errors.push(diag('E_UNREADABLE', { subject: join2(prefix, '.gitignore') }));
      }
    }
    for (const e of entries) {
      if (!isValidUtf8(e.name)) {
        errors.push(diag('E_PATH_ENCODING', { subject: prefix }));
        continue;
      }
      const name = e.name.toString('utf8');
      if (name === '.git') continue;
      const kind = e.isDirectory() ? 'dir' : e.isSymbolicLink() ? 'link' : e.isFile() ? 'file' : null;
      if (kind === null) continue;
      const rel = join2(prefix, name);
      if (isIgnored(rel, kind === 'dir', [...rules, ...configRules])) continue;
      if (kind === 'dir') {
        const childAbs = join(abs, name);
        if (existsSync(join(childAbs, '.git'))) continue;
        walk(childAbs, rel);
      } else {
        found.push({ path: rel, kind });
      }
    }
    rules = before;
  };
  walk(root, '');

  const universe: Universe = { paths: [], kinds: new Map(), onDisk: new Map() };
  for (const { path, kind } of found) {
    if (path === 'docsync.yaml' || path === 'docsync.lock') continue;
    const nfc = path.normalize('NFC');
    if (universe.kinds.has(nfc)) errors.push(diag('E_PATH_COLLISION', { subject: nfc }));
    universe.kinds.set(nfc, kind);
    universe.onDisk.set(nfc, path);
  }
  universe.paths = [...universe.kinds.keys()].sort(comparePaths);
  const lower = new Map<string, string>();
  for (const p of universe.paths) {
    const key = p.toLowerCase();
    const first = lower.get(key);
    if (first !== undefined) errors.push(diag('E_PATH_COLLISION', { subject: first }));
    else lower.set(key, p);
  }
  if (errors.length > 0) throw new Raised(errors);
  return universe;
}
```

Note the order in §7.2 step 2.1/2.2: the spec reads `.gitignore` before listing. Reading it requires knowing it exists, so list first, then load rules, then evaluate entries; the observable result is the same. If listing fails, `.gitignore` is irrelevant because no entry is evaluated.

```ts
import { isAlias, isMap, isPair, isScalar, isSeq, parseAllDocuments, visit, type Node } from 'yaml';

export type YamlValue = string | number | boolean | null | YamlValue[] | YamlMap;
export interface YamlMap { readonly kind: 'map'; readonly entries: Map<string, YamlNodeInfo> }
export interface YamlNodeInfo { value: YamlValue; plainSource: string | null }

class Violation extends Error {}

function convert(node: unknown, text: string): YamlNodeInfo {
  if (isScalar(node)) {
    const plain = node.type === 'PLAIN' && node.range ? text.slice(node.range[0], node.range[1]) : null;
    const v = node.value;
    if (v !== null && !['string', 'number', 'boolean'].includes(typeof v)) throw new Violation();
    return { value: v as string | number | boolean | null, plainSource: plain };
  }
  if (isSeq(node)) return { value: node.items.map((i) => convert(i, text).value), plainSource: null };
  if (isMap(node)) {
    const entries = new Map<string, YamlNodeInfo>();
    for (const pair of node.items) {
      if (!isPair(pair) || !isScalar(pair.key) || typeof pair.key.value !== 'string') {
        throw new Violation();
      }
      if (pair.key.type === 'PLAIN' && pair.key.source !== undefined &&
        typeof pair.key.source !== 'string') throw new Violation();
      if (entries.has(pair.key.value)) throw new Violation();
      entries.set(pair.key.value, convert(pair.value, text));
    }
    return { value: { kind: 'map', entries }, plainSource: null };
  }
  if (node === null || node === undefined) return { value: null, plainSource: null };
  throw new Violation();
}

// SPEC §9.2
export function parseStrictYaml(raw: string): YamlNodeInfo | null {
  const text = raw.startsWith('﻿') ? raw.slice(1) : raw;
  const docs = parseAllDocuments(text, { version: '1.2', schema: 'core', uniqueKeys: true, merge: false });
  if (!Array.isArray(docs) || docs.length !== 1) return null;
  const doc = docs[0]!;
  if (doc.errors.length > 0) return null;
  let bad = false;
  visit(doc, {
    Alias: () => { bad = true; },
    Node: (_k, n: Node) => { if (isAlias(n) || n.anchor || n.tag) bad = true; },
    Pair: (_k, p) => { if (isScalar(p.key) && p.key.value === '<<') bad = true; },
  });
  if (bad) return null;
  try {
    return convert(doc.contents, text);
  } catch (e) {
    if (e instanceof Violation) return null;
    throw e;
  }
}
```

```ts
import { join } from 'node:path';
import { Raised, diag } from '../core/diagnostics.ts';
import { comparePaths } from '../core/order.ts';
import { isRepoPath } from '../core/repo-path.ts';
import type { Binding, Config, Diagnostic } from '../core/types.ts';
import { parsePattern } from '../pattern/parse.ts';
import { parseStrictYaml, type YamlMap, type YamlNodeInfo, type YamlValue } from './yaml-profile.ts';

const isMap = (v: YamlValue): v is YamlMap => typeof v === 'object' && v !== null && !Array.isArray(v);
const isStrings = (v: YamlValue): v is string[] => Array.isArray(v) && v.every((x) => typeof x === 'string');

function readText(root: string): string {
  const path = join(root, 'docsync.yaml');
  try {
    if (!statSync(path).isFile()) throw new Error();
    return readFileSync(path, 'utf8');
  } catch {
    throw new Raised([diag('E_CONFIG_MISSING')]);
  }

// SPEC §9.3
export function readConfig(root: string): { config: Config; attached: Diagnostic[] } {
  const doc = parseStrictYaml(readText(root));
  if (!doc || !isMap(doc.value)) throw new Raised([diag('E_CONFIG')]);
  const top = doc.value.entries;
  const version = top.get('version');
  if (!version || version.plainSource !== '1') throw new Raised([diag('E_CONFIG_VERSION')]);

  const fatal: Diagnostic[] = [];
  const attached: Diagnostic[] = [];
  for (const key of top.keys()) {
    if (!['version', 'gitignore', 'ignore', 'dependents'].includes(key)) {
      fatal.push(diag('E_UNKNOWN_KEY', { subject: key }));
    }
  }
  const gitignore = top.get('gitignore')?.value;
  if (gitignore !== undefined && typeof gitignore !== 'boolean') {
    fatal.push(diag('E_CONFIG', { subject: 'gitignore' }));
  }
  const ignore = top.get('ignore')?.value;
  if (ignore !== undefined && !isStrings(ignore)) fatal.push(diag('E_CONFIG', { subject: 'ignore' }));

  const bindings: Binding[] = [];
  const deps = top.get('dependents')?.value;
  if (deps === undefined || !isMap(deps)) {
    fatal.push(diag('E_CONFIG', { subject: 'dependents' }));
  } else {
    for (const [key, info] of deps.entries) {
      collectBinding(key, info, fatal, attached, bindings);
    }
  }
  if (fatal.length > 0) throw new Raised(fatal);
  bindings.sort((a, b) => comparePaths(a.dependent, b.dependent));
  return {
    config: {
      ignore: isStrings(ignore ?? []) ? (ignore as string[] | undefined) ?? [] : [],
      useGitignore: gitignore === undefined ? true : gitignore === true,
      bindings,
    },
    attached,
  };
}

function collectBinding(
  key: string, info: YamlNodeInfo, fatal: Diagnostic[], attached: Diagnostic[], out: Binding[],
): void {
  if (!isRepoPath(key)) {
    fatal.push(diag('E_CONFIG', { subject: key }));
    return;
  }
  const v = info.value;
  const covers = isMap(v) ? v.entries.get('covers')?.value : undefined;
  if (!isMap(v) || covers === undefined || !isStrings(covers) || covers.length === 0) {
    fatal.push(diag('E_CONFIG', { dependent: key }));
    return;
  }
  for (const k of v.entries.keys()) {
    if (k !== 'covers') fatal.push(diag('E_UNKNOWN_KEY', { dependent: key, subject: k }));
  }
  for (const s of covers) {
    if (!parsePattern(s)) attached.push(diag('E_PATTERN', { dependent: key, subject: s }));
  }
  out.push({ dependent: key, covers });
}
```

```ts
import { createHash } from 'node:crypto';
import { readFileSync, readlinkSync } from 'node:fs';
import { join } from 'node:path';
import { Raised, diag } from '../core/diagnostics.ts';
import type { Universe } from '../universe/walk.ts';

export const sha256Hex = (bytes: Uint8Array): string =>
  createHash('sha256').update(bytes).digest('hex');

function isBinary(bytes: Buffer): boolean {
  return bytes.subarray(0, 8192).includes(0);
}

function crlfToLf(bytes: Buffer): Buffer {
  const out = Buffer.alloc(bytes.length);
  let n = 0;
  for (let i = 0; i < bytes.length; i++) {
    if (bytes[i] === 0x0d && bytes[i + 1] === 0x0a) continue;
    out[n++] = bytes[i]!;
  }
  return out.subarray(0, n);
}

// SPEC §10.2
export function normalizedContent(root: string, u: Universe, path: string): Buffer {
  const abs = join(root, u.onDisk.get(path) ?? path);
  try {
    if (u.kinds.get(path) === 'link') {
      const target = readlinkSync(abs, 'utf8').replaceAll('\\', '/');
      return Buffer.concat([Buffer.from('link\u0000'), Buffer.from(target, 'utf8')]);
    }
    const bytes = readFileSync(abs);
    return Buffer.concat([Buffer.from('file\u0000'), isBinary(bytes) ? bytes : crlfToLf(bytes)]);
  } catch {
    throw new Raised([diag('E_UNREADABLE', { subject: path })]);
  }
}

// SPEC §10.3
export function fileHash(root: string, u: Universe, path: string): string {
  return sha256Hex(normalizedContent(root, u, path));
}

// SPEC §10.4 (pure part: input already hashed)
export function coverHashFrom(entries: ReadonlyArray<[string, string]>): string {
  const parts = entries.map(([path, hash]) => `${path}\u0000${hash}\n`);
  return sha256Hex(Buffer.from(parts.join(''), 'utf8'));
}
```

```ts
import { existsSync, readFileSync, renameSync, writeFileSync } from 'node:fs';
import { join } from 'node:path';
import { parseStrictYaml, type YamlMap, type YamlValue } from '../config/yaml-profile.ts';
import { Raised, diag } from '../core/diagnostics.ts';
import { sortPaths } from '../core/order.ts';
import { quote } from '../core/quote.ts';
import { isRepoPath } from '../core/repo-path.ts';
import type { Lock, LockEntry } from '../core/types.ts';

const LOCK = 'docsync.lock';
const fail = (code: 'E_LOCK' | 'E_LOCK_VERSION') => new Raised([diag(code)]);
const isMap = (v: YamlValue | undefined): v is YamlMap =>
  typeof v === 'object' && v !== null && !Array.isArray(v);

// SPEC §11.1
export function readLock(root: string): Lock {
  const path = join(root, LOCK);
  if (!existsSync(path)) return { entries: new Map() };
  const doc = parseStrictYaml(readFileSync(path, 'utf8'));
  if (!doc || !isMap(doc.value)) throw fail('E_LOCK');
  const top = doc.value.entries;
  if (top.get('version')?.plainSource !== '1') throw fail('E_LOCK_VERSION');
  const keys = [...top.keys()].sort();
  if (keys.join(',') !== 'dependents,version') throw fail('E_LOCK');
  const deps = top.get('dependents')?.value;
  if (!isMap(deps)) throw fail('E_LOCK');
  const entries = new Map<string, LockEntry>();
  for (const [dep, info] of deps.entries) {
    const v = info.value;
    if (!isRepoPath(dep) || !isMap(v) || [...v.entries.keys()].sort().join(',') !== 'covers,hash') {
      throw fail('E_LOCK');
    }
    const covers = v.entries.get('covers')?.value;
    const hash = v.entries.get('hash')?.value;
    const okCovers = Array.isArray(covers) && covers.length > 0 &&
      covers.every((c) => typeof c === 'string');
    if (!okCovers || typeof hash !== 'string' || !/^[0-9a-f]{64}$/u.test(hash)) throw fail('E_LOCK');
    entries.set(dep, { covers: covers as string[], hash });
  }
  return { entries };
}

// SPEC §11.2
export function lockText(lock: Lock): string {
  if (lock.entries.size === 0) return 'version: 1\ndependents: {}\n';
  let out = 'version: 1\ndependents:\n';
  for (const dep of sortPaths([...lock.entries.keys()])) {
    const e = lock.entries.get(dep)!;
    out += `  ${quote(dep)}:\n    covers:\n`;
    for (const c of e.covers) out += `      - ${quote(c)}\n`;
    out += `    hash: ${e.hash}\n`;
  }
  return out;
}

// SPEC §11.3
export function writeLock(root: string, lock: Lock): boolean {
  const path = join(root, LOCK);
  const text = lockText(lock);
  if (existsSync(path)) {
    const current = readFileSync(path).toString('utf8').replaceAll('\r\n', '\n');
    if (current === text) return false;
  }
  const tmp = `${path}.tmp-${process.pid}`;
  writeFileSync(tmp, text, 'utf8');
  renameSync(tmp, path);
  return true;
}
```

```ts
import { Raised, diag, sortDiagnostics } from '../core/diagnostics.ts';
import { sortPaths } from '../core/order.ts';
import type { Binding, Diagnostic, Lock, Reason, Result } from '../core/types.ts';
import { coverHashFrom } from '../hash/hash.ts';
import { patternMatches, select } from '../pattern/match.ts';
import { parsePattern, type ParsedPattern } from '../pattern/parse.ts';

export interface EngineFs {
  isDependentFile(path: string): boolean;
  fileHash(path: string): string;
}

// SPEC §8.5
export function resolveCovers(b: Binding, universe: readonly string[]): string[] {
  const patterns = b.covers.map((s) => parsePattern(s)) as ParsedPattern[];
  const candidates = universe.filter((p) => p !== b.dependent);
  const problems: Diagnostic[] = [];
  patterns.forEach((p, i) => {
    if (!candidates.some((path) => patternMatches(p, path))) {
      problems.push(diag('E_EMPTY_PATTERN', { dependent: b.dependent, subject: b.covers[i] }));
    }
  });
  const covered = select(patterns, candidates);
  if (covered.length === 0) problems.push(diag('E_EMPTY_COVERS', { dependent: b.dependent }));
  if (problems.length > 0) throw new Raised(problems);
  return covered;
}

function coverHash(covered: readonly string[], fs: EngineFs): string {
  const problems: Diagnostic[] = [];
  const entries: Array<[string, string]> = [];
  for (const path of covered) {
    try {
      entries.push([path, fs.fileHash(path)]);
    } catch (e) {
      if (!(e instanceof Raised)) throw e;
      problems.push(...e.diagnostics);
    }
  }
  if (problems.length > 0) throw new Raised(problems);
  return coverHashFrom(entries);
}

const sameList = (a: readonly string[], b: readonly string[]) =>
  a.length === b.length && a.every((x, i) => x === b[i]);

// SPEC §12.1
export function evaluate(
  b: Binding, universe: readonly string[], lock: Lock, attached: readonly Diagnostic[], fs: EngineFs,
): Result {
  const base = { dependent: b.dependent, covers: b.covers };
  const problems: Diagnostic[] = [...attached];
  if (!fs.isDependentFile(b.dependent)) {
    problems.push(diag('E_DEPENDENT_MISSING', { dependent: b.dependent }));
  }
  let covered: string[] = [];
  const collect = (fn: () => void) => {
    try { fn(); } catch (e) {
      if (!(e instanceof Raised)) throw e;
      problems.push(...e.diagnostics);
    }
  };
  if (attached.length === 0) collect(() => { covered = resolveCovers(b, universe); });
  let current = '';
  if (problems.length === 0) collect(() => { current = coverHash(covered, fs); });
  if (problems.length > 0) {
    const diagnostics = sortDiagnostics(problems.map((d) => ({ ...d, dependent: b.dependent })));
    return { ...base, state: 'invalid', reasons: [], covered: [], current: '', diagnostics };
  }
  const reasons: Reason[] = [];
  const entry = lock.entries.get(b.dependent);
  if (!entry) reasons.push('unrecorded');
  else {
    if (!sameList(entry.covers, b.covers)) reasons.push('binding-changed');
    if (entry.hash !== current) reasons.push('content-changed');
  }
  const state = reasons.length > 0 ? 'stale' : 'ok';
  return { ...base, state, reasons, covered, current, diagnostics: [] };
}

// SPEC §12.2 step 5
export function orphans(bindings: readonly Binding[], lock: Lock): Diagnostic[] {
  const bound = new Set(bindings.map((b) => b.dependent));
  return sortPaths([...lock.entries.keys()])
    .filter((d) => !bound.has(d))
    .map((d) => diag('W_ORPHAN', { subject: d }));
}
```

```ts
import { posix } from 'node:path';
import { diag } from '../core/diagnostics.ts';
import { isRepoPath } from '../core/repo-path.ts';
import type { Diagnostic, Result } from '../core/types.ts';

const toPosix = (p: string) => p.replaceAll('\\', '/');

// SPEC §13.4
export function toRepoPath(
  arg: string, cwd: string, root: string, windows = process.platform === 'win32',
): string | null {
  const a = windows ? toPosix(arg) : arg;
  const base = windows ? toPosix(cwd) : cwd;
  const r = windows ? toPosix(root) : root;
  const joined = posix.normalize(posix.isAbsolute(a) || /^[A-Za-z]:\//u.test(a) ? a : `${base}/${a}`);
  const prefix = r.endsWith('/') ? r : `${r}/`;
  if (!joined.startsWith(prefix)) return null;
  const rest = joined.slice(prefix.length).replace(/\/$/u, '').normalize('NFC');
  return isRepoPath(rest) ? rest : null;
}

// SPEC §13.3
export function selectResults(
  args: readonly string[], cwd: string, root: string, results: readonly Result[],
): { selected: Result[]; errors: Diagnostic[] } {
  if (args.length === 0) return { selected: [...results], errors: [] };
  const named = new Set<string>();
  const errors: Diagnostic[] = [];
  const known = new Set(results.map((r) => r.dependent));
  for (const arg of args) {
    const path = toRepoPath(arg, cwd, root);
    if (path === null || !known.has(path)) errors.push(diag('E_UNKNOWN_DEPENDENT', { subject: arg }));
    else named.add(path);
  }
  return { selected: results.filter((r) => named.has(r.dependent)), errors };
}
```

```ts
import { Raised, diag } from '../core/diagnostics.ts';

export type Args =
  | { mode: 'help' } | { mode: 'version' }
  | { mode: 'check'; files: boolean; json: boolean; root?: string; paths: string[] }
  | { mode: 'write'; all: boolean; json: boolean; root?: string; paths: string[] };

export const HELP = `Usage:
  docsync [--files] [--json] [--root <dir>] [<file>...]
      Check that each file's covered files are unchanged since its last review.
  docsync --write [--json] [--root <dir>] (--all | <file>...)
      Record in docsync.lock that you reviewed the named files against their covered files.
  docsync --version | --help
`;

const FLAGS = new Set(['--files', '--json', '--write', '--all', '--version', '--help']);
const usage = (subject: string) => new Raised([diag('E_USAGE', { subject })]);

// SPEC §13.2
export function parseArgs(argv: readonly string[]): Args {
  const seen = new Set<string>();
  const paths: string[] = [];
  let root: string | undefined;
  const mark = (flag: string) => {
    if (seen.has(flag)) throw usage(flag);
    seen.add(flag);
  };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i]!;
    if (a === '--') { paths.push(...argv.slice(i + 1)); break; }
    if (a === '--root' || a.startsWith('--root=')) {
      mark('--root');
      root = a === '--root' ? argv[++i] : a.slice('--root='.length);
      if (root === undefined || root === '') throw usage('--root');
    } else if (FLAGS.has(a)) mark(a);
    else if (a.startsWith('-') && a !== '-') throw usage(a);
    else paths.push(a);
  }
  const total = seen.size + paths.length;
  if (seen.has('--version')) {
    if (total > 1) throw usage('--version');
    return { mode: 'version' };
  }
  if (seen.has('--help')) {
    if (total > 1) throw usage('--help');
    return { mode: 'help' };
  }
  const json = seen.has('--json');
  const opt = root === undefined ? {} : { root };
  if (seen.has('--write')) {
    if (seen.has('--files')) throw usage('--files');
    const all = seen.has('--all');
    if (!all && paths.length === 0) throw usage('--write');
    if (all && paths.length > 0) throw usage('--all');
    return { mode: 'write', all, json, ...opt, paths };
  }
  if (seen.has('--all')) throw usage('--all');
  return { mode: 'check', files: seen.has('--files'), json, ...opt, paths };
}
```

```ts
import { sortDiagnostics } from '../core/diagnostics.ts';
import { quote } from '../core/quote.ts';
import type { Diagnostic, Result } from '../core/types.ts';

export interface JsonDoc {
  mode: 'check' | 'write'; exitCode: number; selected: readonly Result[]; files: boolean;
  diagnostics: readonly Diagnostic[]; written?: ReadonlySet<string>; removed?: readonly string[];
}

type J = null | boolean | number | string | J[] | Array<[string, J]> & { obj: true };
const obj = (pairs: Array<[string, J]>) => Object.assign(pairs, { obj: true as const });

function render(v: J, indent: string): string {
  if (v === null || typeof v === 'boolean' || typeof v === 'number') return String(v);
  if (typeof v === 'string') return quote(v);
  const inner = `${indent}  `;
  if ('obj' in v) {
    if (v.length === 0) return '{}';
    return `{\n${v.map(([k, x]) => `${inner}${quote(k)}: ${render(x, inner)}`).join(',\n')}\n${indent}}`;
  }
  if (v.length === 0) return '[]';
  return `[\n${v.map((x) => `${inner}${render(x as J, inner)}`).join(',\n')}\n${indent}]`;
}

const diagJ = (d: Diagnostic) => obj([
  ['code', d.code], ['severity', d.severity], ['dependent', d.dependent || null],
  ['subject', d.subject || null], ['message', d.message],
]);

// SPEC §14.5
export function jsonText(doc: JsonDoc): string {
  const count = (s: Result['state']) => doc.selected.filter((r) => r.state === s).length;
  const deps = doc.selected.map((r) => {
    const pairs: Array<[string, J]> = [
      ['dependent', r.dependent], ['state', r.state], ['reasons', [...r.reasons]],
      ['covers', [...r.covers]], ['files', doc.files ? [...r.covered] : null],
      ['diagnostics', sortDiagnostics(r.diagnostics).map(diagJ)],
    ];
    if (doc.mode === 'write') pairs.push(['written', doc.written?.has(r.dependent) ?? false]);
    return obj(pairs);
  });
  const top: Array<[string, J]> = [
    ['version', 1], ['mode', doc.mode], ['exitCode', doc.exitCode],
    ['summary', obj([['ok', count('ok')], ['stale', count('stale')], ['invalid', count('invalid')]])],
    ['dependents', deps], ['diagnostics', sortDiagnostics(doc.diagnostics).map(diagJ)],
  ];
  if (doc.mode === 'write') top.push(['removed', [...(doc.removed ?? [])]]);
  return `${render(obj(top), '')}\n`;
}
```

```ts
import { readLock, writeLock } from '../lock/lock.ts';
import { jsonText } from '../report/json.ts';
import { checkText, diagnosticsText, writeText } from '../report/text.ts';
import { computeUniverse, determineRoot } from '../universe/walk.ts';
import { HELP, parseArgs, type Args } from './args.ts';
import { selectResults } from './paths.ts';

export interface Io {
  stdout(s: string): void;
  stderr(s: string): void;
}

declare const VERSION: string | undefined;

// SPEC §9.4: exact-name match in the parent's listing
function isDependentFile(root: string, path: string): boolean {
  const abs = join(root, path);
  try {
    const names = readdirSync(dirname(abs)).map((n) => n.normalize('NFC'));
    return names.includes(basename(path)) && statSync(abs).isFile() &&
      !statSync(abs, { throwIfNoEntry: false })?.isSymbolicLink();
  } catch {
    return false;
  }
}

interface Evaluated { results: Result[]; lock: Lock; global: Diagnostic[] }

// SPEC §12.2
function evaluateAll(root: string, policy: 'strict' | 'discard-invalid'): Evaluated {
  const { config, attached } = readConfig(root);
  const universe = computeUniverse(root, config);
  let lock: Lock;
  try {
    lock = readLock(root);
  } catch (e) {
    if (!(e instanceof Raised) || policy === 'strict') throw e;
    lock = { entries: new Map() };
  }
  const fs: EngineFs = {
    isDependentFile: (p) => isDependentFile(root, p),
    fileHash: (p) => fileHash(root, universe, p),
  };
  const results = config.bindings.map((b) =>
    evaluate(b, universe.paths, lock, attached.filter((d) => d.dependent === b.dependent), fs));
  return { results, lock, global: orphans(config.bindings, lock) };
}

const hasError = (ds: readonly Diagnostic[]) => ds.some((d) => d.severity === 'error');

function emit(io: Io, args: Args & { json: boolean }, mode: 'check' | 'write', exitCode: number,
  selected: Result[], global: Diagnostic[], text: string,
  extra: { written?: Set<string>; removed?: string[] } = {}): number {
  const files = args.mode === 'check' && args.files;
  if (args.json) {
    io.stdout(jsonText({ mode, exitCode, selected, files, diagnostics: global, ...extra }));
  } else {
    io.stdout(text);
    io.stderr(diagnosticsText([...global, ...selected.flatMap((r) => r.diagnostics)]));
  }
  return exitCode;
}

// SPEC §13.5 / §13.6
export function run(argv: readonly string[], cwd: string, io: Io): number {
  let args: Args;
  try {
    args = parseArgs(argv);
  } catch (e) {
    if (!(e instanceof Raised)) throw e;
    io.stderr(diagnosticsText(e.diagnostics));
    return 2;
  }
  if (args.mode === 'help') { io.stdout(HELP); return 0; }
  if (args.mode === 'version') { io.stdout(`${typeof VERSION === 'string' ? VERSION : '0.0.0'}\n`); return 0; }
  const mode = args.mode;
  try {
    const root = determineRoot(cwd, args.root);
    const ev = evaluateAll(root, mode === 'write' && args.all ? 'discard-invalid' : 'strict');
    if (mode === 'check') {
      const { selected, errors } = selectResults(args.paths, cwd, root, ev.results);
      const global = [...ev.global, ...errors];
      const code = hasError(global) || selected.some((r) => r.state === 'invalid') ? 2
        : selected.some((r) => r.state === 'stale') ? 1 : 0;
      return emit(io, args, 'check', code, selected, global,
        checkText(selected, args.files, args.root));
    }
    const sel = args.all ? { selected: ev.results, errors: [] } :
      selectResults(args.paths, cwd, root, ev.results);
    const global = [...ev.global, ...sel.errors];
    if (hasError(global) || sel.selected.some((r) => r.state === 'invalid')) {
      return emit(io, args, 'write', 2, sel.selected, global, checkText(sel.selected, false, args.root),
        { written: new Set(), removed: [] });
    }
    const entries = new Map(ev.lock.entries);
    for (const t of sel.selected) entries.set(t.dependent, { covers: t.covers, hash: t.current });
    const bound = new Set(ev.results.map((r) => r.dependent));
    const removed = [...entries.keys()].filter((d) => !bound.has(d));
    for (const d of removed) entries.delete(d);
    writeLock(root, { entries });
    const written = sel.selected.map((r) => r.dependent);
    return emit(io, args, 'write', 0, sel.selected, global, writeText(written, removed),
      { written: new Set(written), removed });
  } catch (e) {
    if (!(e instanceof Raised)) throw e;
    return emit(io, args, mode, 2, [], e.diagnostics, checkText([], false));
  }
}

export { diag };
```

```ts
const BIN = resolve('dist/index.js');
const docsync = (cwd: string, ...args: string[]) => {
  const r = spawnSync(process.execPath, [BIN, ...args], { cwd, encoding: 'utf8' });
  return { code: r.status, out: r.stdout, err: r.stderr };
};
const CONFIG = 'version: 1\ndependents:\n  CLAUDE.md:\n    covers: [src/**]\n';
const repo = () => makeTree({ 'docsync.yaml': CONFIG, 'CLAUDE.md': '# doc\n', 'src/a.ts': 'a\n' });

describe('§13 workflow', () => {
  it('unrecorded -> write -> ok -> edit -> stale -> write -> ok', () => {
    const root = repo();
    expect(docsync(root).code).toBe(1);
    expect(docsync(root, '--write', 'CLAUDE.md')).toMatchObject({ code: 0, out: 'written  CLAUDE.md\n' });
    expect(docsync(root)).toMatchObject({ code: 0, out: '1 ok, 0 stale, 0 invalid\n' });
    appendFileSync(join(root, 'src/a.ts'), 'b\n');
    const stale = docsync(root);
    expect(stale.code).toBe(1);
    expect(stale.out).toContain('STALE    CLAUDE.md  (content-changed)');
    expect(docsync(root, '--write', 'CLAUDE.md').code).toBe(0);
    expect(docsync(root).code).toBe(0);
  });

  it('rename of a covered file is stale', () => {
    const root = repo();
    docsync(root, '--write', '--all');
    renameSync(join(root, 'src/a.ts'), join(root, 'src/b.ts'));
    expect(docsync(root).code).toBe(1);
  });

  it('editing the dependent itself does not make it stale', () => {
    const root = repo();
    docsync(root, '--write', '--all');
    appendFileSync(join(root, 'CLAUDE.md'), 'more\n');
    expect(docsync(root).code).toBe(0);
  });

  it('runs from a subdirectory and resolves file args against cwd', () => {
    const root = repo();
    expect(docsync(join(root, 'src'), '--write', '../CLAUDE.md').code).toBe(0);
  });

  it('missing dependent is exit 2, others still evaluated', () => {
    const root = makeTree({
      'docsync.yaml': `${CONFIG}  GONE.md:\n    covers: [src/**]\n`,
      'CLAUDE.md': '', 'src/a.ts': '',
    });
    const r = docsync(root, '--json');
    expect(r.code).toBe(2);
    const doc = JSON.parse(r.out);
    expect(doc.dependents.map((d: { state: string }) => d.state)).toEqual(['stale', 'invalid']);
  });

  it('write refuses on lock conflict markers and writes nothing', () => {
    const root = repo();

    const r = docsync(root, '--write', 'CLAUDE.md');
    expect(r.code).toBe(2);
    expect(r.err).toContain('E_LOCK');

  });

  it('--write --all recovers from a corrupt lock', () => {

    writeFileSync(join(root, 'docsync.lock'), 'garbage: [');
    expect(docsync(root, '--write', '--all').code).toBe(0);

    expect(r.code).toBe(2);
    expect(JSON.parse(r.out).diagnostics[0].code).toBe('E_CONFIG_MISSING');
  });

  it('output is deterministic across runs', () => {
    const root = repo();

  });
```
